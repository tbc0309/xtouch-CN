cmake_minimum_required(VERSION 3.22)
option(XTOUCH_FEATURE_XIAOMI_BLE "Include Xiaomi BLE thermometer" ON)
option(XTOUCH_P4_C6_PREINITIALIZED_BT "Use the board's preinitialized C6 Bluetooth controller" ON)
include("${CMAKE_CURRENT_LIST_DIR}/../common/project_version.cmake")
if(CMAKE_HOST_WIN32)
  set(CMAKE_NINJA_FORCE_RESPONSE_FILE ON CACHE BOOL "Use response files for long Windows SDK command lines" FORCE)
endif()
set(XTOUCH_BOARD "esp32-p4-wifi6-touch-lcd-4.3")
set(IDF_TARGET "esp32p4")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(XTOUCH_FEATURE_XIAOMI_BLE)
 if(XTOUCH_FEATURE_BLE_PROVISIONING)
  message(FATAL_ERROR "Xiaomi BLE and BLE provisioning are mutually exclusive")
 endif()
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../common/sdkconfig.xiaomi_ble.defaults")
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.xiaomi_ble.defaults")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../main" "${CMAKE_CURRENT_LIST_DIR}/../.deps/lvgl" "${CMAKE_CURRENT_LIST_DIR}/lp_touch")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main)
project(xtouch_waveshare)

# Keep application and full-install bootloader on the board's validated cache policy.
# Existing sdkconfig values take precedence over defaults: reject stale builds.
if(NOT CONFIG_CACHE_L2_CACHE_LINE_SIZE EQUAL 64)
 message(FATAL_ERROR "P4 requires 64-byte L2 cache lines. Regenerate sdkconfig from this board's defaults before building or using OTA.")
endif()
